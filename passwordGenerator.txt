# ---------------------------------------------------------------------------
# genera_password(lunghezza, maiuscole, numeri, simboli)
#
# Genera una password casuale.
#
# Parametri:
#   lunghezza : int  — numero di caratteri
#   maiuscole : bool — includere lettere maiuscole
#   numeri    : bool — includere cifre
#   simboli   : bool — includere simboli speciali
#
# Requisiti:
#   - La password deve contenere almeno un carattere per ogni
#     categoria abilitata. Se maiuscole=True, ci deve essere
#     almeno una maiuscola. Stesso principio per numeri e simboli.
#   - Il resto dei caratteri viene estratto casualmente dal pool
#     completo delle categorie abilitate.
#   - L'ordine dei caratteri deve essere casuale — non tutti i
#     caratteri garantiti in testa.
#
# Moduli utili:
#   import random
#   import string
#   string.ascii_lowercase  -> "abcdefghijklmnopqrstuvwxyz"
#   string.ascii_uppercase  -> "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
#   string.digits           -> "0123456789"
#   random.choice(seq)      -> elemento casuale da una sequenza
#   random.shuffle(lista)   -> mescola la lista in place (non restituisce nulla)
#   "".join(lista)          -> costruisce una stringa da una lista di caratteri
#
# Restituisce: stringa — la password generata
# ---------------------------------------------------------------------------
#
# ---------------------------------------------------------------------------
# analizza_composizione(password)
#
# Conta quanti caratteri di ogni tipo contiene la password.
#
# Restituisce un dizionario con le chiavi:
#   lunghezza, minuscole, maiuscole, numeri, simboli, altri
#
# Spunto:
#   sum(1 for c in password if c in string.digits)
#   conta quante cifre ci sono nella stringa.
# ---------------------------------------------------------------------------
#
# ---------------------------------------------------------------------------
# calcola_punteggio(password)
#
# Calcola un punteggio di forza tra 0 e 100.
#
# Il punteggio deve tenere conto di almeno questi fattori:
#   - lunghezza della password
#   - presenza di maiuscole
#   - presenza di numeri
#   - presenza di simboli
#   - varieta' complessiva dei caratteri
#
# Come pesare questi fattori e' una tua scelta.
# Definisci le costanti in cima al file con nomi espliciti —
# cosi' chi legge il tuo codice capisce le tue decisioni.
#
# Restituisce: int tra 0 e 100
# ---------------------------------------------------------------------------
#
# ---------------------------------------------------------------------------
# descrivi_punteggio(punteggio)
#
# Restituisce una descrizione testuale del punteggio.
#
# Restituisce una tupla: (etichetta, consiglio)
#   etichetta : stringa descrittiva della forza
#   consiglio : suggerimento su come migliorare la password
#
# Definisci almeno 4 livelli di forza con soglie esplicite come costanti.
# ---------------------------------------------------------------------------
